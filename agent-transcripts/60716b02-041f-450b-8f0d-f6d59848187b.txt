```
- airline: carrier name
```

## 17. Error Handling Module (`error_handler.py`)

```python
from enum import Enum
from dataclasses import dataclass
from typing import Optional, Callable

class ErrorSeverity(Enum):
    INFO = "info"
    WARNING = "warning"
    ERROR = "error"
    CRITICAL = "critical"

class ErrorCode(Enum):
    # API Errors
    API_BAD_JSON = "E001"
    API_RATE_LIMIT = "E002"
    API_TOKEN_LIMIT = "E003"
    API_TIMEOUT = "E004"
    API_INVALID_KEY = "E005"
    
    # Input Errors
    INPUT_BLURRY = "E101"
    INPUT_WRONG_LANG = "E102"
    INPUT_MULTI_FLIGHT = "E103"
    INPUT_MISSING_FIELDS = "E104"
    
    # File Errors
    FILE_CORRUPTED = "E201"
    FILE_WRONG_COLUMNS = "E202"
    FILE_LOCKED = "E203"
    FILE_ENCODING = "E204"
    
    # Parse Errors
    PARSE_DATE = "E301"
    PARSE_AIRLINE = "E302"
    PARSE_PRICE = "E303"
    PARSE_TIME = "E304"
    
    # Structure Errors
    STRUCT_NO_HEADERS = "E401"
    STRUCT_DUPLICATE_BLOCKS = "E402"
    STRUCT_SIMILAR_NAMES = "E403"

@dataclass
class AppError:
    code: ErrorCode
    severity: ErrorSeverity
    message: str
    user_message: str
    recovery_action: Optional[Callable] = None
    retry_allowed: bool = False
    data: dict = None

class ErrorHandler:
    def __init__(self):
        self.errors: list[AppError] = []
        self.warnings: list[AppError] = []
    
    def handle(self, error: AppError) -> bool:
        """Handle error and return True if recovered"""
        if error.severity == ErrorSeverity.WARNING:
            self.warnings.append(error)
            return True  # Continue
        
        self.errors.append(error)
        
        if error.recovery_action:
            try:
                error.recovery_action()
                return True
            except:
                return False
        
        return False
    
    def get_user_message(self, code: ErrorCode) -> str:
        """Return user-friendly error message"""
        messages = {
            ErrorCode.API_BAD_JSON: "The AI response was unclear. Retrying...",
            ErrorCode.API_RATE_LIMIT: "Too many requests. Please wait {seconds} seconds.",
            ErrorCode.API_TIMEOUT: "Request timed out. Click to retry.",
            ErrorCode.INPUT_BLURRY: "Image quality is low. Results may be inaccurate.",
            ErrorCode.FILE_CORRUPTED: "This file appears to be corrupted. Please upload a valid CSV.",
            ErrorCode.FILE_WRONG_COLUMNS: "Column structure doesn't match. Please map columns.",
            # ... etc
        }
        return messages.get(code, "An unexpected error occurred.")
```

## 18. Retry Logic (`retry.py`)

```python
import asyncio
from functools import wraps
from typing import TypeVar, Callable

T = TypeVar('T')

def with_retry(
    max_attempts: int = 3,
    delay_seconds: float = 1.0,
    backoff_factor: float = 2.0,
    exceptions: tuple = (Exception,)
):
    """Decorator for retry logic with exponential backoff"""
    def decorator(func: Callable[..., T]) -> Callable[..., T]:
        @wraps(func)
        async def wrapper(*args, **kwargs) -> T:
            last_exception = None
            delay = delay_seconds
            
            for attempt in range(max_attempts):
                try:
                    return await func(*args, **kwargs)
                except exceptions as e:
                    last_exception = e
                    if attempt < max_attempts - 1:
                        await asyncio.sleep(delay)
                        delay *= backoff_factor
            
            raise last_exception
        return wrapper
    return decorator

# Usage in parser.py:
@with_retry(max_attempts=3, exceptions=(JSONDecodeError, APIError))
async def extract_flight_data(content, api_key, model):
    # ... API call logic
    pass
```

## 4. Gemini Parser Service (`gemini-api.js`)

```javascript
// src/services/gemini-api.js
import { GoogleGenerativeAI } from '@google/generative-ai';

export class GeminiParser {
  constructor(apiKey, model = 'gemini-1.5-flash') {
    this.genAI = new GoogleGenerativeAI(apiKey);
    this.model = this.genAI.getGenerativeModel({ model });
  }

  async extractFlightData(input, isImage = false) {
    const prompt = this.buildPrompt();
    
    let result;
    if (isImage) {
      const imagePart = {
        inlineData: {
          data: input.base64,
          mimeType: input.mimeType
        }
      };
      result = await this.model.generateContent([prompt, imagePart]);
    } else {
      result = await this.model.generateContent(`${prompt}\n\nInput:\n${input}`);
    }

    const response = result.response.text();
    return this.parseResponse(response);
  }

  buildPrompt() {
    return `Extract flight information. Return JSON array:
{
  "dates": {"value": "DD/MM/YY - DD/MM/YY", "confidence": 0.0-1.0},
  "outbound": {"value": "HH:MM CITY (CODE) -> HH:MM CITY (CODE) FLIGHT#", "confidence": 0.0-1.0},
  "inbound": {"value": "same format", "confidence": 0.0-1.0},
  "airline": {"value": "Name", "confidence": 0.0-1.0},
  "destination": {"value": "City (CODE)", "confidence": 0.0-1.0},
  "seats": {"value": number, "confidence": 0.0-1.0},
  "price": {"value": "123$", "confidence": 0.0-1.0}
}

Rules:
- If arrival time < departure time, add (+1) to arrival
- Always use DD/MM/YY format for dates
- Include confidence scores
- If multiple flights, return array`;
  }

  parseResponse(response) {
    // Extract JSON from response
    const jsonMatch = response.match(/\[[\s\S]*\]|\{[\s\S]*\}/);
    if (!jsonMatch) throw new Error('No JSON found in response');
    
    const data = JSON.parse(jsonMatch[0]);
    return Array.isArray(data) ? data : [data];
  }

  // Fix overnight flights
  fixOvernightTime(timeString) {
    const match = timeString.match(/(\d{2}:\d{2}).*->.*(\d{2}:\d{2})/);
    if (!match) return timeString;
    
    const [_, dep, arr] = match;
    if (arr < dep && !timeString.includes('(+1)')) {
      return timeString.replace(arr, `${arr}(+1)`);
    }
    return timeString;
  }
}
```

    """Normalize airline name to standard format."""

def fix_overnight_time(time_string: str) -> str:
    """Add (+1) to arrival time if it's earlier than departure (overnight flight)."""
    if not time_string or '(+1)' in time_string:
        return time_string
    
    # Extract times from format like "23:00 TLV -> 02:30 VIE W6 123"
    time_pattern = r'(\d{2}:\d{2}).*?->\s*(\d{2}:\d{2})'
    match = re.search(time_pattern, time_string)
    
    if match:
        dep_time = match.group(1)
        arr_time = match.group(2)
        
        # Compare times (simple string comparison works for HH:MM format)
        if arr_time < dep_time:
            # Insert (+1) after arrival time
            return time_string.replace(arr_time, f"{arr_time}(+1)", 1)
    
    return time_string


def parse_date_for_sorting(date_str: str) -> datetime:
    """Parse date string for sorting. Returns datetime object."""
    if not date_str or pd.isna(date_str):
        return datetime.max
    
    # Extract first date from "DD/MM - DD/MM" or "DD/MM/YY - DD/MM/YY"
    match = re.search(r'(\d{1,2})/(\d{1,2})(?:/(\d{2,4}))?', str(date_str))
    if match:
        day = int(match.group(1))
        month = int(match.group(2))
        year = int(match.group(3)) if match.group(3) else datetime.now().year
        if year < 100:
            year += 2000
        try:
            return datetime(year, month, day)
        except ValueError:
            pass
    
    return datetime.max


def read_inventory(file) -> pd.DataFrame:

def identify_airline_blocks(df: pd.DataFrame) -> dict:
    """
    Identify airline blocks in the dataframe.
    Returns dict: {airline_name: {'start': row_index, 'end': row_index}}
    """
    blocks = {}
    current_airline = None
    current_start = None
    
    for idx, row in df.iterrows():
        first_col = str(row.iloc[0]).strip() if pd.notna(row.iloc[0]) else ""
        
        # Check if this is an airline header row (only first column has value, others empty)
        other_cols_empty = all(pd.isna(row.iloc[1:]) or str(row.iloc[1:].values[0]).strip() == "")
        
        if first_col and other_cols_empty and first_col not in ['', 'nan']:
            # This might be an airline header
            # Save previous block
            if current_airline and current_start is not None:
                blocks[current_airline] = {'start': current_start, 'end': idx - 1}
            
            current_airline = first_col
            current_start = idx
        elif current_airline and first_col:
            # This is a data row within a block
            pass
    
    # Save last block
    if current_airline and current_start is not None:
        blocks[current_airline] = {'start': current_start, 'end': len(df) - 1}
    
    return blocks

    """
    dates_col = COLUMNS['dates']
    outbound_col = COLUMNS['outbound']
    
    new_dates = str(new_data.get('dates', '')).strip()
    new_outbound = str(new_data.get('outbound', '')).strip()
    
    for idx, row in df.iterrows():
        existing_dates = str(row.get(dates_col, '')).strip()
        existing_outbound = str(row.get(outbound_col, '')).strip()
        
        if existing_dates == new_dates and existing_outbound == new_outbound:
            return idx
    
    return -1


def update_inventory(df: pd.DataFrame, new_data: dict) -> pd.DataFrame:
    """
    Update inventory with new flight data.
    - Finds correct airline block
    - Checks for duplicates
    - Inserts and sorts within block
    """
    df = df.copy()
    airline = normalize_airline(new_data.get('airline', 'Other'))
    
    # Create new row
    new_row = {
        COLUMNS['dates']: new_data.get('dates', ''),
        COLUMNS['outbound']: fix_overnight_time(new_data.get('outbound', '')),
        COLUMNS['inbound']: fix_overnight_time(new_data.get('inbound', '')),
        COLUMNS['airline']: airline,
        COLUMNS['destination']: new_data.get('destination', ''),
        COLUMNS['seats']: new_data.get('seats', ''),
        COLUMNS['price']: new_data.get('price', '')
    }
    
    # Check for duplicate
    dup_idx = find_duplicate(df, new_data)
    if dup_idx >= 0:
        # Update existing row (price and seats)
        df.at[dup_idx, COLUMNS['seats']] = new_row[COLUMNS['seats']]
        df.at[dup_idx, COLUMNS['price']] = new_row[COLUMNS['price']]
        return df, "updated", dup_idx
    
    # Find airline blocks
    blocks = identify_airline_blocks(df)
    
    if airline in blocks:
        # Insert into existing block
        block = blocks[airline]
        block_start = block['start'] + 1  # After header
        block_end = block['end']
        
        # Get block data (excluding header)
        block_df = df.iloc[block_start:block_end + 1].copy()
        
        # Add new row
        new_row_df = pd.DataFrame([new_row])
        block_df = pd.concat([block_df, new_row_df], ignore_index=True)
        
        # Sort block by date
        block_df['_sort_date'] = block_df[COLUMNS['dates']].apply(parse_date_for_sorting)
        block_df = block_df.sort_values('_sort_date').drop('_sort_date', axis=1)
        
        # Reconstruct dataframe
        before = df.iloc[:block_start]
        after = df.iloc[block_end + 1:]
        df = pd.concat([before, block_df, after], ignore_index=True)
        
        return df, "added", -1
    else:
        # Create new airline block at the end
        # Add empty row, then airline header, then data
        empty_row = pd.DataFrame([{col: '' for col in df.columns}])
        header_row = pd.DataFrame([{df.columns[0]: airline, **{col: '' for col in df.columns[1:]}}])
        data_row = pd.DataFrame([new_row])
        
        df = pd.concat([df, empty_row, header_row, data_row], ignore_index=True)
        
        return df, "added (new block)", -1


def export_csv(df: pd.DataFrame) -> bytes:
    """Export dataframe to CSV bytes with UTF-8-SIG encoding (for Hebrew)."""
    buffer = io.BytesIO()
    df.to_csv(buffer, index=False, encoding='utf-8-sig')
    return buffer.getvalue()

def extract_flight_data_with_gemini(content, api_key: str, is_image: bool = False) -> dict:
    """Extract flight data using Google Gemini API."""
    if not GEMINI_AVAILABLE:
        st.error("google-generativeai package not installed. Run: pip install google-generativeai")
        return None
    
    try:
        genai.configure(api_key=api_key)
        model = genai.GenerativeModel('gemini-1.5-flash')
        
        prompt = """Extract flight information from the following input. 
Return ONLY a valid JSON object with these exact fields:
{
    "dates": "DD/MM - DD/MM or DD/MM/YY - DD/MM/YY format",
    "outbound": "HH:MM CITY (CODE) -> HH:MM CITY (CODE) FLIGHT#",
    "inbound": "HH:MM CITY (CODE) -> HH:MM CITY (CODE) FLIGHT#",
    "airline": "Airline Name",
    "destination": "City (CODE)",
    "seats": number,
    "price": "price with currency symbol"
}

If any field is not found, use empty string or 0 for seats.
Return ONLY the JSON, no other text."""

        if is_image:
            # Content is PIL Image
            response = model.generate_content([prompt, content])
        else:
            # Content is text
            response = model.generate_content(f"{prompt}\n\nInput:\n{content}")
        
        # Extract JSON from response
        response_text = response.text
        
        # Try to find JSON in response
        json_match = re.search(r'\{[^{}]*\}', response_text, re.DOTALL)
        if json_match:
            return json.loads(json_match.group())
        
        # Try parsing entire response as JSON
        return json.loads(response_text)
        
    except json.JSONDecodeError as e:
        st.error(f"Failed to parse LLM response as JSON: {e}")
        st.code(response_text if 'response_text' in dir() else "No response")
        return None
    except Exception as e:
        st.error(f"Gemini API error: {e}")
        return None

def normalize_airline(name: str) -> str:
    if not name:
        return "Other"
    return AIRLINE_ALIASES.get(name.lower().strip(), name.strip())


def fix_overnight_time(time_string: str) -> str:
    if not time_string or '(+1)' in time_string:
        return time_string
    match = re.search(r'(\d{2}:\d{2}).*?->\s*(\d{2}:\d{2})', time_string)
    if match and match.group(2) < match.group(1):
        return time_string.replace(match.group(2), f"{match.group(2)}(+1)", 1)
    return time_string


def parse_date_for_sorting(date_str: str) -> datetime:
    if not date_str or pd.isna(date_str):
        return datetime.max
    match = re.search(r'(\d{1,2})/(\d{1,2})(?:/(\d{2,4}))?', str(date_str))
    if match:
        day, month = int(match.group(1)), int(match.group(2))
        year = int(match.group(3)) if match.group(3) else datetime.now().year
        if year < 100:
            year += 2000
        try:
            return datetime(year, month, day)
        except ValueError:
            pass
    return datetime.max


def read_inventory(file) -> pd.DataFrame:
    try:
        if file.name.endswith(('.xlsx', '.xls')):
            df = pd.read_excel(file)
        else:
            for encoding in ['utf-8-sig', 'utf-8', 'cp1255']:
                try:
                    file.seek(0)
                    df = pd.read_csv(file, encoding=encoding)
                    break
                except:
                    continue
            else:
                return None
        for col in COLUMN_ORDER:
            if col not in df.columns:
                df[col] = ""
        return df
    except Exception as e:
        st.error(f"Error: {e}")
        return None


def identify_airline_blocks(df: pd.DataFrame) -> dict:
    blocks = {}
    current_airline = None
    current_start = None
    
    for idx, row in df.iterrows():
        first_col = str(row.iloc[0]).strip() if pd.notna(row.iloc[0]) else ""
        other_empty = all(pd.isna(v) or str(v).strip() == "" for v in row.iloc[1:])
        
        if first_col and other_empty and first_col not in ['', 'nan']:
            if current_airline and current_start is not None:
                blocks[current_airline] = {'start': current_start, 'end': idx - 1}
            current_airline = first_col
            current_start = idx
    
    if current_airline and current_start is not None:
        blocks[current_airline] = {'start': current_start, 'end': len(df) - 1}
    
    return blocks


def find_duplicate(df: pd.DataFrame, new_data: dict) -> int:
    new_dates = str(new_data.get('dates', '')).strip()
    new_outbound = str(new_data.get('outbound', '')).strip()
    
    for idx, row in df.iterrows():
        if (str(row.get(COLUMNS['dates'], '')).strip() == new_dates and 
            str(row.get(COLUMNS['outbound'], '')).strip() == new_outbound):
            return idx
    return -1


def update_inventory(df: pd.DataFrame, new_data: dict):
    df = df.copy()
    airline = normalize_airline(new_data.get('airline', 'Other'))
    
    new_row = {
        COLUMNS['dates']: new_data.get('dates', ''),
        COLUMNS['outbound']: fix_overnight_time(new_data.get('outbound', '')),
        COLUMNS['inbound']: fix_overnight_time(new_data.get('inbound', '')),
        COLUMNS['airline']: airline,
        COLUMNS['destination']: new_data.get('destination', ''),
        COLUMNS['seats']: new_data.get('seats', ''),
        COLUMNS['price']: new_data.get('price', '')
    }
    
    dup_idx = find_duplicate(df, new_data)
    if dup_idx >= 0:
        df.at[dup_idx, COLUMNS['seats']] = new_row[COLUMNS['seats']]
        df.at[dup_idx, COLUMNS['price']] = new_row[COLUMNS['price']]
        return df, "updated", dup_idx
    
    blocks = identify_airline_blocks(df)
    
    if airline in blocks:
        block = blocks[airline]
        block_start = block['start'] + 1
        block_end = block['end']
        block_df = df.iloc[block_start:block_end + 1].copy()
        new_row_df = pd.DataFrame([new_row])
        block_df = pd.concat([block_df, new_row_df], ignore_index=True)
        block_df['_sort'] = block_df[COLUMNS['dates']].apply(parse_date_for_sorting)
        block_df = block_df.sort_values('_sort').drop('_sort', axis=1)
        before = df.iloc[:block_start]
        after = df.iloc[block_end + 1:]
        df = pd.concat([before, block_df, after], ignore_index=True)
        return df, "added", -1
    else:
        empty = pd.DataFrame([{col: '' for col in df.columns}])
        header = pd.DataFrame([{df.columns[0]: airline, **{c: '' for c in df.columns[1:]}}])
        data = pd.DataFrame([new_row])
        df = pd.concat([df, empty, header, data], ignore_index=True)
        return df, "added (new block)", -1

def extract_with_gemini(content, api_key: str, is_image: bool = False) -> dict:
    if not GEMINI_AVAILABLE:
        st.error("google-genai not installed. Run: pip install google-genai")
        return None
    
    try:
        client = genai.Client(api_key=api_key)
        
        prompt = """Extract flight information. Return ONLY valid JSON:
{
    "dates": "DD/MM - DD/MM",
    "outbound": "HH:MM CITY (CODE) -> HH:MM CITY (CODE) FLIGHT#",
    "inbound": "HH:MM CITY (CODE) -> HH:MM CITY (CODE) FLIGHT#",
    "airline": "Airline Name",
    "destination": "City (CODE)",
    "seats": 1,
    "price": "199$"
}"""

        if is_image:
            # Convert PIL Image to bytes
            img_byte_arr = io.BytesIO()
            content.save(img_byte_arr, format='PNG')
            img_bytes = img_byte_arr.getvalue()
            
            response = client.models.generate_content(
                model="gemini-2.0-flash",
                contents=[prompt, {"mime_type": "image/png", "data": img_bytes}]
            )
        else:
            response = client.models.generate_content(
                model="gemini-2.0-flash",
                contents=f"{prompt}\n\nInput:\n{content}"
            )
        
        text = response.text
        json_match = re.search(r'\{[^{}]*\}', text, re.DOTALL)
        if json_match:
            return json.loads(json_match.group())
        return json.loads(text)
        
    except Exception as e:
        st.error(f"Gemini error: {e}")
        return None

def normalize_airline(name: str) -> str:
    if not name:
        return "Other"
    return AIRLINE_ALIASES.get(name.lower().strip(), name.strip())


def fix_overnight_time(time_string: str) -> str:
    if not time_string or '(+1)' in time_string:
        return time_string
    match = re.search(r'(\d{2}:\d{2}).*?->\s*(\d{2}:\d{2})', str(time_string))
    if match and match.group(2) < match.group(1):
        return time_string.replace(match.group(2), f"{match.group(2)}(+1)", 1)
    return time_string


def parse_date_for_sorting(date_str: str) -> datetime:
    if not date_str or pd.isna(date_str):
        return datetime.max
    match = re.search(r'(\d{1,2})/(\d{1,2})(?:/(\d{2,4}))?', str(date_str))
    if match:
        day, month = int(match.group(1)), int(match.group(2))
        year = int(match.group(3)) if match.group(3) else datetime.now().year
        if year < 100:
            year += 2000
        try:
            return datetime(year, month, day)
        except ValueError:
            pass
    return datetime.max


def read_inventory(file) -> pd.DataFrame:
    try:
        if file.name.endswith(('.xlsx', '.xls')):
            df = pd.read_excel(file)
        else:
            for encoding in ['utf-8-sig', 'utf-8', 'cp1255']:
                try:
                    file.seek(0)
                    df = pd.read_csv(file, encoding=encoding)
                    break
                except:
                    continue
            else:
                return None
        return df
    except Exception as e:
        st.error(f"שגיאה: {e}")
        return None


def get_stats(df: pd.DataFrame) -> dict:
    """Calculate dashboard statistics."""
    if df is None or df.empty:
        return {'flights': 0, 'seats': 0, 'airlines': 0}
    
    # Count non-header rows (rows with actual flight data)
    data_rows = df[df[COLUMNS['dates']].notna() & (df[COLUMNS['dates']] != '')]
    data_rows = data_rows[~data_rows[COLUMNS['outbound']].isna() | (data_rows[COLUMNS['outbound']] != '')]
    
    flights = len(data_rows)
    
    # Sum seats
    seats = 0
    for val in data_rows[COLUMNS['seats']]:
        try:
            seats += int(val)
        except:
            pass
    
    # Count unique airlines
    airlines = data_rows[COLUMNS['airline']].dropna().nunique()
    
    return {'flights': flights, 'seats': seats, 'airlines': airlines}


def filter_dataframe(df: pd.DataFrame, search: str, airline_filter: str) -> pd.DataFrame:
    """Filter dataframe by search term and airline."""
    if df is None:
        return None
    
    filtered = df.copy()
    
    if search:
        mask = filtered.apply(lambda row: search.lower() in str(row.values).lower(), axis=1)
        filtered = filtered[mask]
    
    if airline_filter and airline_filter != "הכל":
        filtered = filtered[filtered[COLUMNS['airline']] == airline_filter]
    
    return filtered


def identify_airline_blocks(df: pd.DataFrame) -> dict:
    blocks = {}
    current_airline = None
    current_start = None
    
    for idx, row in df.iterrows():
        first_col = str(row.iloc[0]).strip() if pd.notna(row.iloc[0]) else ""
        other_empty = all(pd.isna(v) or str(v).strip() == "" for v in row.iloc[1:])
        
        if first_col and other_empty and first_col not in ['', 'nan']:
            if current_airline and current_start is not None:
                blocks[current_airline] = {'start': current_start, 'end': idx - 1}
            current_airline = first_col
            current_start = idx
    
    if current_airline and current_start is not None:
        blocks[current_airline] = {'start': current_start, 'end': len(df) - 1}
    
    return blocks


def find_duplicate(df: pd.DataFrame, new_data: dict) -> int:
    new_dates = str(new_data.get('dates', '')).strip()
    new_outbound = str(new_data.get('outbound', '')).strip()
    
    for idx, row in df.iterrows():
        if (str(row.get(COLUMNS['dates'], '')).strip() == new_dates and 
            str(row.get(COLUMNS['outbound'], '')).strip() == new_outbound):
            return idx
    return -1


def update_inventory(df: pd.DataFrame, new_data: dict):
    df = df.copy()
    airline = normalize_airline(new_data.get('airline', 'Other'))
    
    new_row = {
        COLUMNS['dates']: new_data.get('dates', ''),
        COLUMNS['outbound']: fix_overnight_time(new_data.get('outbound', '')),
        COLUMNS['inbound']: fix_overnight_time(new_data.get('inbound', '')),
        COLUMNS['airline']: airline,
        COLUMNS['destination']: new_data.get('destination', ''),
        COLUMNS['seats']: new_data.get('seats', ''),
        COLUMNS['price']: new_data.get('price', '')
    }
    
    dup_idx = find_duplicate(df, new_data)
    if dup_idx >= 0:
        df.at[dup_idx, COLUMNS['seats']] = new_row[COLUMNS['seats']]
        df.at[dup_idx, COLUMNS['price']] = new_row[COLUMNS['price']]
        return df, "עודכן", dup_idx
    
    blocks = identify_airline_blocks(df)
    
    if airline in blocks:
        block = blocks[airline]
        block_start = block['start'] + 1
        block_end = block['end']
        block_df = df.iloc[block_start:block_end + 1].copy()
        new_row_df = pd.DataFrame([new_row])
        block_df = pd.concat([block_df, new_row_df], ignore_index=True)
        block_df['_sort'] = block_df[COLUMNS['dates']].apply(parse_date_for_sorting)
        block_df = block_df.sort_values('_sort').drop('_sort', axis=1)
        before = df.iloc[:block_start]
        after = df.iloc[block_end + 1:]
        df = pd.concat([before, block_df, after], ignore_index=True)
        return df, "נוסף", -1
    else:
        empty = pd.DataFrame([{col: '' for col in df.columns}])
        header = pd.DataFrame([{df.columns[0]: airline, **{c: '' for c in df.columns[1:]}}])
        data = pd.DataFrame([new_row])
        df = pd.concat([df, empty, header, data], ignore_index=True)
        return df, "נוסף (בלוק חדש)", -1

def extract_with_gemini(content, api_key: str, is_image: bool = False) -> dict:
    if not GEMINI_AVAILABLE:
        st.error("google-genai לא מותקן. הרץ: pip install google-genai")
        return None
    
    try:
        client = genai.Client(api_key=api_key)
        
        prompt = """Extract flight information from this input. Return ONLY valid JSON:
{
    "dates": "DD/MM - DD/MM",
    "outbound": "HH:MM CITY (CODE) -> HH:MM CITY (CODE) FLIGHT#",
    "inbound": "HH:MM CITY (CODE) -> HH:MM CITY (CODE) FLIGHT#",
    "airline": "Airline Name",
    "destination": "City (CODE)",
    "seats": 1,
    "price": "$199"
}
If arrival time < departure time, add (+1) after arrival time."""

        if is_image:
            img_byte_arr = io.BytesIO()
            content.save(img_byte_arr, format='PNG')
            img_bytes = img_byte_arr.getvalue()
            
            response = client.models.generate_content(
                model="gemini-2.0-flash",
                contents=[prompt, {"mime_type": "image/png", "data": img_bytes}]
            )
        else:
            response = client.models.generate_content(
                model="gemini-2.0-flash",
                contents=f"{prompt}\n\nInput:\n{content}"
            )
        
        text = response.text
        json_match = re.search(r'\{[^{}]*\}', text, re.DOTALL)
        if json_match:
            return json.loads(json_match.group())
        return json.loads(text)
        
    except Exception as e:
        st.error(f"שגיאת Gemini: {e}")
        return None
